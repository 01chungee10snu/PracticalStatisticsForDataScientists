예제 1: 기본 계산 및 출력
# 기본 계산 예제 x = 10 y = 20 result = x + y print(f"x = {x}, y = {y}") print(f"x + y = {result}") # 추가 계산 average = (x + y) / 2 print(f"평균: {average}")
예제 2: 기술통계량 계산
# 기술통계량 계산 예제 import statistics import math # 학생 점수 데이터 scores = [85, 90, 78, 92, 88, 76, 95, 89, 84, 91] # 기술통계량 계산 mean_score = statistics.mean(scores) median_score = statistics.median(scores) stdev_score = statistics.stdev(scores) print("=== 기술통계량 분석 ===") print(f"데이터: {scores}") print(f"평균: {mean_score:.2f}") print(f"중앙값: {median_score}") print(f"표준편차: {stdev_score:.2f}") print(f"최솟값: {min(scores)}") print(f"최댓값: {max(scores)}")
예제 3: 데이터 분석 (시뮬레이션)
# 데이터 분석 시뮬레이션 import random # 랜덤 데이터 생성 random.seed(42) data = [random.randint(60, 100) for _ in range(20)] # 데이터 분석 total = sum(data) count = len(data) average = total / count # 등급 분류 grades = {'A': 0, 'B': 0, 'C': 0, 'D': 0, 'F': 0} for score in data: if score >= 90: grades['A'] += 1 elif score >= 80: grades['B'] += 1 elif score >= 70: grades['C'] += 1 elif score >= 60: grades['D'] += 1 else: grades['F'] += 1 print("=== 성적 분석 결과 ===") print(f"총 학생 수: {count}명") print(f"평균 점수: {average:.1f}점") print(f"등급별 분포: {grades}")
